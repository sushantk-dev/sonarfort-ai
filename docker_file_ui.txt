# ── STAGE 1: Build Angular Application ──────────────────────────────────────
FROM us-docker.pkg.dev/iaas-gcr-reg-prd-ad3d/golden/node:20-alpine3.21 AS build-stage

WORKDIR /app

# Force npm to install ALL dependencies (including devDependencies)
ENV NODE_ENV=development

# Copy package configurations first to leverage Docker layer caching
COPY package*.json ./

USER root
# Disable strict SSL to bypass corporate proxy issues
RUN npm config set strict-ssl false

# Install global Angular CLI
RUN npm install -g @angular/cli

# Clean install dependencies ignoring any production-only flags
RUN npm install

# Copy the rest of your Angular source code securely on a single line
COPY . .

# Build using npm lifecycle script directly while bypassing style budget limits
RUN npm run build -- --configuration=production

# ── STAGE 2: Serve Bundle via Nginx ─────────────────────────────────────────

FROM us-docker.pkg.dev/iaas-gcr-reg-prd-ad3d/golden/nginx:alpine
USER root
# Clean default Nginx assets
RUN rm -rf /usr/share/nginx/html/* /etc/nginx/conf.d/default.conf

RUN chmod -R 777 /var/cache/nginx /var/run /var/log/nginx && chown -R efx_container_user:efx_container_user /var/cache/nginx /var/run /var/log/nginx

RUN sed -i 's|/var/run/nginx.pid|/tmp/nginx.pid|g' /etc/nginx/nginx.conf

COPY --chown=efx_container_user:efx_container_user --from=build-stage /app/dist/sonarfort-ai/browser /usr/share/nginx/html

RUN printf 'server {\n\
    listen 8080;\n\
    server_name _;\n\
    root /usr/share/nginx/html;\n\
    index index.html;\n\
\n\
    # SSO: id_tokens with many group claims can exceed the default 8k header limit\n\
    large_client_header_buffers 4 32k;\n\
\n\
    location /api/ {\n\
        proxy_pass http://localhost:8000/api/;\n\
        proxy_http_version 1.1;\n\
        proxy_set_header Host $host;\n\
        proxy_set_header X-Real-IP $remote_addr;\n\
        proxy_read_timeout 300s;\n\
        client_max_body_size 50M;\n\
    }\n\
\n\
    location /fortify/ {\n\
        proxy_pass http://localhost:8001/;\n\
        proxy_http_version 1.1;\n\
        proxy_set_header Host $host;\n\
        proxy_set_header X-Real-IP $remote_addr;\n\
        proxy_read_timeout 300s;\n\
    }\n\
\n\
    # Health checks hit public (no-SSO-token) endpoints\n\
    location = /health/sonar   { proxy_pass http://localhost:8000/api/health; access_log off; }\n\
    location = /health/fortify { proxy_pass http://localhost:8001/health; access_log off; }\n\
\n\
    # 1. Cache rules for static assets (CSS, JS, Images, Fonts)\n\
    location ~* \\.(?:css|js|woff2?|svg|gif|jpe?g|png)$ {\n\
        root /usr/share/nginx/html;\n\
        expires 1y;\n\
        add_header Cache-Control "public";\n\
        access_log off;\n\
    }\n\
\n\
    # 2. Merged fallback for SPA routing with caching disabled (covers /login and /callback)\n\
    location / {\n\
        try_files $uri $uri/ /index.html;\n\
        add_header Cache-Control "no-store, no-cache, must-revalidate, proxy-revalidate, max-age=0";\n\
    }\n\
}\n' > /etc/nginx/conf.d/sonarfort.conf

USER efx_container_user

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
